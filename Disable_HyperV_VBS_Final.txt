# ------------------------------------------------------------
# Helper Function: Write-Log (timestamped IT logging)
# ------------------------------------------------------------
function Write-Log {
    param(
        [string]$Message,
        [string]$Level = "INFO",
        [string]$Color = "White"
    )
    $timestamp = (Get-Date).ToString("yyyy-MM-dd HH:mm:ss")
    $logEntry = "[$timestamp] [$Level] $Message"
    if ($script:logFile) {
        try {
            Add-Content -Path $script:logFile -Value $logEntry -Force -ErrorAction Stop
        } catch {
            Write-Host "[LOG WRITE ERROR] Could not write to log: $($_.Exception.Message)" -ForegroundColor Red
        }
    }

    if ($Level -eq "SUCCESS") {
        Write-Host $Message -ForegroundColor Green
    } elseif ($Level -eq "WARN") {
        Write-Host $Message -ForegroundColor Yellow
    } elseif ($Level -eq "ERROR") {
        Write-Host $Message -ForegroundColor Red
    } else {
        Write-Host $Message -ForegroundColor $Color
    }
}

# ------------------------------------------------------------
# Helper: ExecuteCommandAndLog (from DG Readiness Tool)
# Uses cmd /c instead of Invoke-Expression to avoid injection risk
# ------------------------------------------------------------
function ExecuteCommandAndLog {
    param([string]$_cmd)
    try {
        Write-Log "Executing: $_cmd" -Level "INFO"
        $CmdOutput = cmd /c $_cmd 2>&1 | Out-String
        Write-Log "Output: $CmdOutput" -Level "INFO"
    } catch {
        Write-Log "Exception while executing $_cmd : $($_.Exception.Message)" -Level "WARN"
    }
}

# Auto-Elevation: If running in standard PowerShell, prompt for Administrator rights
# ------------------------------------------------------------

    Write-Host "Administrator privileges required." -ForegroundColor Yellow
    Write-Host "Requesting Administrator permission via UAC prompt..." -ForegroundColor Cyan

    $rawPath = $PSCommandPath
    if (-not $rawPath) {
        $rawPath = $MyInvocation.MyCommand.Path
    }
    if (-not $rawPath -and $MyInvocation.MyCommand.Name) {
        $candidate = Join-Path -Path $PSScriptRoot -ChildPath $MyInvocation.MyCommand.Name
        if (Test-Path -Path $candidate) {
            $rawPath = $candidate
        }
    }

    $scriptPath = $null
    if ($rawPath -and (Test-Path -Path $rawPath)) {
        $scriptPath = (Resolve-Path -Path $rawPath).Path
    }

    if ($scriptPath) {
        try {
            if ($scriptPath.EndsWith(".txt", [System.StringComparison]::OrdinalIgnoreCase)) {
                Start-Process -FilePath "powershell.exe" -ArgumentList "-NoExit -NoProfile -ExecutionPolicy Bypass -Command `"& { [string]`$c = Get-Content -Raw -LiteralPath '$scriptPath'; Invoke-Expression `$c }`"" -Verb RunAs
            } else {
                Start-Process -FilePath "powershell.exe" -ArgumentList "-NoExit -NoProfile -ExecutionPolicy Bypass -File `"$scriptPath`" -LogDir `"$LogDir`"" -Verb RunAs
            }
            Write-Host "Elevated Administrator session launched in a separate window." -ForegroundColor Green
            exit
        } catch {
            Write-Host "Administrator elevation was declined or cancelled." -ForegroundColor Red
            exit
        }
    } else {
        Write-Host "ERROR: Could not resolve script file path for auto-elevation." -ForegroundColor Red
        Write-Host "Please right-click PowerShell, choose 'Run as administrator', and run the script file." -ForegroundColor Yellow
        exit
    }
}

# Step 4: Disabling VBS in Registry (Set to 0 AND Delete keys)
#         Combines both approaches: Set-ItemProperty + REG DELETE
# ------------------------------------------------------------
Write-Log "`n=== Step 4: Disabling Virtualization-Based Security (VBS) in Registry ===" -Level "INFO" -Color Cyan

# 4a: Set values to 0 first (keeps keys for audit, immediate effect)

    $dgPath = "HKLM:\SYSTEM\CurrentControlSet\Control\DeviceGuard"
    if (-not (Test-Path -Path $dgPath)) { New-Item -Path $dgPath -Force | Out-Null }
    Set-ItemProperty -Path $dgPath -Name "EnableVirtualizationBasedSecurity" -Value 0 -Type DWord -Force
    Set-ItemProperty -Path $dgPath -Name "RequirePlatformSecurityFeatures" -Value 0 -Type DWord -Force
    Set-ItemProperty -Path $dgPath -Name "Locked" -Value 0 -Type DWord -Force
    Write-Log "VBS Registry values set to 0: EnableVirtualizationBasedSecurity, RequirePlatformSecurityFeatures, Locked" -Level "SUCCESS"
} catch {
    Write-Log "Error setting VBS registry values to 0: $($_.Exception.Message)" -Level "ERROR"

# Windows 11 24H2 is always Redstone+ (build > 10586); simplified from DG Readiness Tool

    $CmdOutput = DISM.EXE /Online /disable-Feature /FeatureName:Microsoft-Hyper-V-Hypervisor /NoRestart | Out-String
    if (-not $CmdOutput.Contains("The operation completed successfully.")) {
        $CmdOutput = DISM.EXE /Online /disable-Feature /FeatureName:Microsoft-Hyper-V-Online /NoRestart | Out-String
    }
    Write-Log "DISM output: $CmdOutput" -Level "INFO"
    if ($CmdOutput.Contains("The operation completed successfully.")) {
        Write-Log "Disabling Hyper-V and IOMMU via DISM successful." -Level "SUCCESS"
    } else {
        Write-Log "DISM Hyper-V disable did not report success (may already be disabled). Check log." -Level "WARN"
    }
} catch {
    Write-Log "DISM Hyper-V disable error: $($_.Exception.Message)" -Level "WARN"
}

# ------------------------------------------------------------
# Step 8: Configuring UEFI Opt-Out via built-in SecConfig.efi
# ------------------------------------------------------------
Write-Log "`n=== Step 8: Configuring UEFI Opt-Out via built-in SecConfig.efi ===" -Level "INFO" -Color Cyan
$secConfigSrc = "$env:windir\System32\SecConfig.efi"
if (Test-Path -Path $secConfigSrc) {
    $freeDrive = $null
    $efiMounted = $false
    try {
        # Pick the first available drive letter from Z: down to E: to avoid collisions
        $availableLetters = 90..69 | ForEach-Object { [char]$_ + ":" } | Where-Object { -not (Test-Path -Path $_) }
        if (-not $availableLetters) {
            Write-Log "No available drive letter found to mount EFI partition." -Level "ERROR"
            throw "No free drive letter available."
        }
        $freeDrive = $availableLetters[0]

        Write-Log "Mounting EFI System Partition to temporary drive $freeDrive..." -Level "INFO"
        mountvol $freeDrive /s
        $efiMounted = $true

        $efiBootDir = "$freeDrive\EFI\Microsoft\Boot"
        if (-not (Test-Path -Path $efiBootDir)) {
            New-Item -ItemType Directory -Path $efiBootDir -Force | Out-Null
        }
        Copy-Item -Path $secConfigSrc -Destination "$efiBootDir\SecConfig.efi" -Force

        # Verify the copy with hash check — prevents boot failure from corrupt SecConfig.efi
        $srcHash = (Get-FileHash -Path $secConfigSrc -Algorithm SHA256).Hash
        $dstHash = (Get-FileHash -Path "$efiBootDir\SecConfig.efi" -Algorithm SHA256).Hash
        if ($srcHash -ne $dstHash) {
            Write-Log "ERROR: SecConfig.efi copy verification FAILED (hash mismatch). Aborting Step 8 to prevent boot corruption." -Level "ERROR"
            Write-Log "  Source hash: $srcHash" -Level "ERROR"
            Write-Log "  Dest hash:   $dstHash" -Level "ERROR"
            throw "SecConfig.efi copy is corrupt — refusing to set BCD bootsequence."
        }
        Write-Log "SecConfig.efi copied & verified (SHA256 match). Safe to set BCD." -Level "SUCCESS"

        # Only set BCD bootsequence AFTER file copy is verified — atomic operation
        cmd /c 'bcdedit /create "{0cb3b571-2f2e-4343-a879-d86a476d7215}" /d "DGOptOut" /application osloader' 2>$null
        bcdedit /set '{0cb3b571-2f2e-4343-a879-d86a476d7215}' path "\EFI\Microsoft\Boot\SecConfig.efi" | Out-Null
        bcdedit /set '{0cb3b571-2f2e-4343-a879-d86a476d7215}' loadoptions "DISABLE-LSA-ISO,DISABLE-VBS" | Out-Null
        bcdedit /set '{0cb3b571-2f2e-4343-a879-d86a476d7215}' device "partition=$freeDrive" | Out-Null
        # Set bootsequence LAST — so if power dies before this line, no boot entry points to anything
        bcdedit /set '{bootmgr}' bootsequence '{0cb3b571-2f2e-4343-a879-d86a476d7215}' | Out-Null

        Write-Log "UEFI Opt-Out successfully configured in BCD bootsequence (atomic: file verified before BCD)." -Level "SUCCESS"
    } catch {
        Write-Log "Warning configuring SecConfig.efi: $($_.Exception.Message)" -Level "WARN"
    } finally {
        if ($efiMounted -and $freeDrive) {
            mountvol $freeDrive /d 2>$null
            Write-Log "EFI System Partition dismounted from $freeDrive." -Level "INFO"
        }
    }
} else {
    Write-Log "SecConfig.efi not found in System32; skipping UEFI variable deletion." -Level "WARN"
}

# ------------------------------------------------------------
# Step 9: Verifying hypervisor launch type in BCD
# ------------------------------------------------------------
Write-Log "`n=== Step 9: Verifying hypervisor launch type in BCD ===" -Level "INFO" -Color Cyan
try {
    $bcdCheck = (bcdedit /enum '{current}' 2>&1 | Select-String -Pattern "hypervisorlaunchtype" | Out-String).Trim()
    if ($bcdCheck -match 'hypervisorlaunchtype\s+Off') {
        Write-Log "BCD verified: hypervisorlaunchtype = Off" -Level "SUCCESS"
    } elseif ($bcdCheck) {
        # Value exists but is not Off - re-apply and re-verify
        Write-Log "hypervisorlaunchtype is NOT Off: $bcdCheck" -Level "WARN"
        $null = bcdedit /set hypervisorlaunchtype off
        $recheck = (bcdedit /enum '{current}' 2>&1 | Select-String -Pattern "hypervisorlaunchtype" | Out-String).Trim()
        if ($recheck -match 'hypervisorlaunchtype\s+Off') {
            Write-Log "Re-applied successfully: hypervisorlaunchtype = Off" -Level "SUCCESS"
        } else {
            Write-Log "Could not confirm hypervisorlaunchtype=Off after re-apply: $recheck" -Level "ERROR"
        }
    } else {
        # Entry not present - default can be Auto; set it explicitly
        Write-Log "hypervisorlaunchtype entry not found in BCD; setting it explicitly to Off." -Level "WARN"
        $finalBcd = (bcdedit /set hypervisorlaunchtype off 2>&1 | Out-String).Trim()
        Write-Log "hypervisorlaunchtype set to Off ($finalBcd)" -Level "SUCCESS"
    }
} catch {
    Write-Log "BCD verification note: $($_.Exception.Message)" -Level "WARN"
}

    Write-Log "Fast Startup disabled in registry." -Level "SUCCESS"
} catch {
    Write-Log "Could not disable Fast Startup: $($_.Exception.Message)" -Level "WARN"
}

# ------------------------------------------------------------
# Post-Execution Verification: Check if VBS is actually OFF right now
# ------------------------------------------------------------
Write-Log "`n=== Post-Execution Verification: Checking if VBS is truly OFF ===" -Level "INFO" -Color Cyan
$vbsStillOn = $false
try {
    $dg = Get-CimInstance -ClassName Win32_DeviceGuard -Namespace root\Microsoft\Windows\DeviceGuard -ErrorAction Stop
    Write-Log "VirtualizationBasedSecurityStatus: $($dg.VirtualizationBasedSecurityStatus) (0=Off, 1=Pending, 2=Running)" -Level "INFO" -Color Gray
    Write-Log "SecurityServicesRunning: $($dg.SecurityServicesRunning -join ', ') (empty = good)" -Level "INFO" -Color Gray
    Write-Log "SecurityServicesConfigured: $($dg.SecurityServicesConfigured -join ', ')" -Level "INFO" -Color Gray
    if ($dg.VirtualizationBasedSecurityStatus -eq 2 -or ($dg.SecurityServicesRunning -and $dg.SecurityServicesRunning.Count -gt 0)) {
        $vbsStillOn = $true
        Write-Log "WARNING: VBS is still RUNNING — hypervisor will hold VT-x. Reboot required + check OEM bloatware." -Level "WARN"
    } else {
        Write-Log "VBS is OFF (or pending reboot to turn off). Good." -Level "SUCCESS"
    }
} catch {
    Write-Log "Could not query Win32_DeviceGuard: $($_.Exception.Message)" -Level "WARN"
}

# Check HVCI registry
try {
    $hvci = Get-ItemProperty -Path "HKLM:\SYSTEM\CurrentControlSet\Control\DeviceGuard\Scenarios\HypervisorEnforcedCodeIntegrity" -ErrorAction SilentlyContinue
    if ($hvci -and $hvci.Enabled -eq 1) {
        Write-Log "WARNING: HVCI registry still shows Enabled=1 — OEM bloatware may re-enable on reboot." -Level "WARN"
    } else {
        Write-Log "HVCI registry is OFF (or key absent). Good." -Level "SUCCESS"
    }
} catch {
    Write-Log "HVCI registry check note: $($_.Exception.Message)" -Level "WARN"
}

if ($vbsStillOn) {
    Write-Host ""
    Write-Host "============================================================" -ForegroundColor Yellow
    Write-Host "  VBS IS STILL RUNNING — REBOOT REQUIRED" -ForegroundColor Yellow
    Write-Host "============================================================" -ForegroundColor Yellow
    Write-Host "  VBS did not turn off yet. After reboot:" -ForegroundColor White
    Write-Host "  1. Re-open Windows Security -> Device Security -> Core Isolation" -ForegroundColor White
    Write-Host "  2. Confirm Memory Integrity is OFF (turn it OFF manually if ON)" -ForegroundColor White
    Write-Host "  3. Press F3 (or Fn+F3) during boot if prompted" -ForegroundColor White
    Write-Host "  4. Re-run this script's verification block" -ForegroundColor White
    Write-Host "============================================================" -ForegroundColor Yellow
} else {
    Write-Host ""
    Write-Host "VBS is OFF. Reboot to finalize, then EVE-NG should get VT-x." -ForegroundColor Green
}
